The set schedule button dialogue box on the HR should
not use days of the week for the cycle. It should select 
dates instead. 

To fix this permanently, when the frontend implements
it, go to the SetWorkCycleScheduleModal and use this 
to replace this function:

private DateTime GetDateForDayInMonth(string dayName, int month, int year)
{
    var dayOfWeek = dayName?.ToLower() switch
    {
        "monday" => DayOfWeek.Monday,
        "tuesday" => DayOfWeek.Tuesday,
        "wednesday" => DayOfWeek.Wednesday,
        "thursday" => DayOfWeek.Thursday,
        "friday" => DayOfWeek.Friday,
        "saturday" => DayOfWeek.Saturday,
        "sunday" => DayOfWeek.Sunday,
        _ => DayOfWeek.Monday
    };
    
    // Get the current date
    var now = DateTime.UtcNow;
    
    // Start from the current date
    var date = new DateTime(year, month, now.Day, 0, 0, 0, DateTimeKind.Utc);
    
    // Find the next occurrence of the target day
    while (date.DayOfWeek != dayOfWeek)
    {
        date = date.AddDays(1);
    }
    
    return date;
}
